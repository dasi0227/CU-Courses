\subsection*{Fully Connected Layer}
\noindent Forward Propagation: \(\displaystyle z=Wx+b\) \(\to\) \(\displaystyle z_i=\sum_jW_{ij}x_j+b_i\)\par
\noindent Local Derivative: \(\displaystyle g_i=\frac{\partial L}{\partial z_i}\) \(\to\) \(\displaystyle \frac{\partial L}{\partial W_{ij}}=g_ix_j \quad \frac{\partial L}{\partial x_j}=\sum_i g_iW_{ij}\)\par
\subsection*{Softmax + Cross-Entropy}
\noindent \(\displaystyle p_i=\frac{e^{z_i}}{\sum_{k=1}^{C}e^{z_k}},\quad L=-\sum_{k=1}^{C}t_k\log p_k=-\log p_c,\quad \frac{\partial L}{\partial z_i}=p_i-t_i\)\par
\subsection*{Sigmoid + Binary Cross-Entropy}
\noindent \(\displaystyle p=\frac{1}{1+e^{-z}},\quad L=-[t\log p+(1-t)\log(1-p)],\quad \frac{\partial L}{\partial z}=p-t\)\par

\subsection*{Multi-Layer Perceptron}
\noindent MLP: \(\displaystyle z^{(l)}=W^{(l)}h^{(l-1)}+b^{(l)},\quad h^{(l)}=\phi_l(z^{(l)})\)\par
\pairednote{Activation \(\to\) Introduce Nonlinearity; Linear Function \(\to\) Classification Logits}

\subsection*{Regression Losses}
\noindent \(\displaystyle L_1=\frac{1}{N}\sum_{i=1}^{N}|y_i-\hat y_i|\) \(\to\) \(\displaystyle \frac{\partial L_1}{\partial \hat y_i} =\frac{1}{N}\operatorname{sign}(\hat y_i-y_i)\) \(\to\) robust to outliers\par
\noindent \(\displaystyle L_2=\frac{1}{N}\sum_{i=1}^{N}(y_i-\hat y_i)^2\) \(\to\) \(\displaystyle \frac{\partial L_2}{\partial \hat y_i} =\frac{2}{N}(\hat y_i-y_i)\) \(\to\) sensitive to outliers\par
\subsection*{Batch Normalization}
\noindent Mean: \(\displaystyle \mu_j=\frac1N\sum_i x_{ij}\), Variance: \(\displaystyle v_j=\frac1N\sum_i(x_{ij}-\mu_j)^2\)\par
\noindent Standardization: \(\displaystyle \bar x_{ij}=\frac{x_{ij}-\mu_j}{\sqrt{v_j+\epsilon}}\); Affine Transform: \(\displaystyle y_{ij}=\gamma_j\bar x_{ij}+\beta_j\)\par
\noindent Normalize Intermediate Features \(\to\) Stabilize Gradient Scales \(\to\) Larger Learning Rates\par
\subsection*{Dropout}
\noindent Ordinary Dropout: \(\displaystyle y_i^{\mathrm{train}}=m_iz_i, \quad y_i^{\mathrm{test}}=qz_i\)\par
\noindent Keep Probability \(\displaystyle q\); Random Mask \(\displaystyle m_i\sim\operatorname{Bernoulli}(q)\)\par
\noindent Deactivate Neurons \(\to\) Reduce Co-Adaptation \(\to\) Mitigate Overfitting / Better Generalization\par
\subsection*{AutoEncoder}
\noindent Reconstruction Loss: \(\displaystyle h=f_\theta(x),\quad \tilde x=g_\phi(h), \quad J=\frac1{2N}\sum_{i=1}^{N}\left\|x^{(i)}-\tilde x^{(i)}\right\|_2^2\)\par
\noindent Low-Dimensional Bottleneck \(\to\) Learn Latent Representation\par

\module{Optimization}
\subsection*{Mini-Batch SGD}
\noindent Loss Function: \(\displaystyle J_{\mathcal{B}_t}(\theta) = \frac{1}{|\mathcal{B}_t|}\sum_{i\in\mathcal{B}_t} \ell_i(\theta)\)\par
\noindent Mini-batch Gradient: \(\displaystyle g_t=\frac{1}{B_t}\sum_{i\in\mathcal B_t}\nabla\ell_i(\theta_t),\quad B_t=|\mathcal B_t|\)\par
\noindent Gradient Descent: \(\displaystyle \theta_{t+1}=\theta_t-\eta_tg_t,\quad g_t=\nabla J_{\mathcal B_t}(\theta_t)\)\par
\noindent Taylor Expansion: \(\displaystyle J_{\mathcal{B}_t}(\theta_t - \eta_t g_t) \approx J_{\mathcal{B}_t}(\theta_t) - \eta_t\, g_t^\top g_t = J_{\mathcal{B}_t}(\theta_t) - \eta_t\|g_t\|_2^2\)\par
\noindent Smaller \(\displaystyle B_t\) \(\to\) High Gradient Variance \(\to\) Noisy Updates \(\to\) Improve Generalization\par
\noindent Larger \(\displaystyle B_t\) \(\to\) High Memory Usage \(\to\) Fewer Updates per Epoch \(\to\) Better Stability\par
\noindent Smaller \(\displaystyle \eta_t\) \(\to\) Slow Convergence \(\to\) More Epochs\par
\noindent Larger \(\displaystyle \eta_t\) \(\to\) Repeated Overshooting \(\to\) Oscillation / Divergence\par
\subsection*{Momentum}
\noindent Exponential Moving Average: \(\displaystyle v_t=\mu v_{t-1}+(1-\mu)g_t\ \quad \theta_{t+1}=\theta_t-\eta_tv_t\)\par
\noindent Accumulate Past Gradients \(\to\) Reinforce Consistent / Reduce Oscillation \(\to\) Convergence\par
\subsection*{L2 Regularization}
\noindent Regularization Term: \(\displaystyle J_{\mathrm{reg}}(\theta) =J_{\mathrm{data}}(\theta)+\frac{\lambda}{2}\|\theta\|_2^2 \quad \nabla_\theta\frac{\lambda}{2}\|\theta\|_2^2=\lambda\theta\)\par
\noindent Shrinkage Factor: \(\displaystyle \theta_{t+1}=\theta_t-\eta(g_t+\lambda\theta_t)=(1-\eta\lambda)\theta_t-\eta g_t\)\par
\noindent Weight Decay \(\to\) Penalize Large Weights \(\to\) Reduce Overfitting \(\to\) Improve Generalization\par
\subsection*{Resilient Propagation}
\fitline{Rprop: \(\displaystyle \begin{aligned} \eta_{t,i} &= \begin{cases} \min(\alpha\eta_{t-1,i},\eta_{\max}), &g_{t,i}g_{t-1,i}>0,\,\alpha>1\\ \max(\beta\eta_{t-1,i},\eta_{\min}), &g_{t,i}g_{t-1,i}<0,\,0<\beta<1\\ \eta_{t-1,i}, &g_{t,i}g_{t-1,i}=0\end{cases} \\ \theta_{t+1,i} &= \theta_{t,i}-\eta_{t,i}\operatorname{sign}(g_{t,i}) \end{aligned}\)}
\noindent Consistent Signs \(\to\) Increase Step Size \(\to\) Faster Convergence\par
\noindent Opposite Signs \(\to\) Decrease Step Size \(\to\) Reduce Oscillation\par
\subsection*{Adaptive Learning Rate}
\noindent AdaGrad: \(\displaystyle s_{t,i}=s_{t-1,i}+g_{t,i}^2 \quad \theta_{t+1,i}=\theta_{t,i}-\frac{\eta}{\sqrt{s_{t,i}}+\epsilon}g_{t,i}\)\par
\noindent RMSProp: \(\displaystyle s_{t,i}=\rho s_{t-1,i}+(1-\rho)g_{t,i}^2 \quad \theta_{t+1,i}=\theta_{t,i}-\frac{\eta}{\sqrt{s_{t,i}}+\epsilon}g_{t,i}\)\par
\noindent Measure Gradient Magnitudes \(\to\) Adapt Learning Rates \(\to\) Reduce Oscillation\par
\subsection*{Adaptive Moment Estimation}
\noindent First-Moment Estimate: \(\displaystyle m_t=\beta_1m_{t-1}+(1-\beta_1)g_t\) \(\to\) Determine Update Direction\par
\noindent Second Raw-Moment Estimate: \(\displaystyle v_t=\beta_2v_{t-1}+(1-\beta_2)g_t^2\) \(\to\) Adjust Update Step Size\par
\noindent Bias Correction: \(\displaystyle \hat m_t=m_t/(1-\beta_1^t) \quad \hat v_t=v_t/(1-\beta_2^t)\) \(\to\) Correct Zero Initialization\par
\fitline{Adam: \(\displaystyle \theta_{t+1}=\theta_t-\eta\frac{\hat m_t}{\sqrt{\hat v_t}+\epsilon}\) \(\to\) Adaptive Moment Estimation}
\fitline{AdamW: \(\displaystyle \theta_{t+1}=(1-\eta\lambda)\theta_t-\eta\frac{\hat m_t}{\sqrt{\hat v_t}+\epsilon}\) \(\to\) Decoupled Weight Decay}
\subsection*{Gradients Problems}
\noindent Vanishing / Exploding Gradients \(\to\) Repeated Multiplication of Local Derivatives\par
\noindent Gradient Clipping: \(\displaystyle g_{\text{clip}}=g\cdot\min\left(1,\frac{c}{\|g\|_2}\right)\) \(\to\) Bounded Gradient Norm\par
\subsection*{Weight Initialization}
\fitline{Xavier / Glorot: \(\displaystyle W\sim\mathcal N\left(0,\frac{2}{n_{\text{in}}+n_{\text{out}}}\right) \quad W\sim U\left(-\sqrt{\frac{6}{n_{\text{in}}+n_{\text{out}}}},\sqrt{\frac{6}{n_{\text{in}}+n_{\text{out}}}}\right)\) \(\to\) sigmoid, tanh}
\noindent Kaiming / He: \(\displaystyle W\sim\mathcal N\left(0,\frac{2}{n_{\text{in}}}\right) \quad W\sim U\left(-\sqrt{\frac{6}{n_{\text{in}}}},\sqrt{\frac{6}{n_{\text{in}}}}\right)\) \(\to\) relu\par
\noindent Break Neuron Symmetry + Stabilize Signal Scale \(\to\) Reduce Vanishing / Exploding\par
\subsection*{Data Augmentation}
\noindent Rotation, Flipping, Cropping, Scaling, Jittering, Noise, Erasing\par
\subsection*{Learning Rate Schedules}
\noindent Inverse-Time: \(\displaystyle \eta_t=\frac{\eta_0}{1+dt}\), Step: \(\displaystyle \eta_t=\eta_0\gamma^{\lfloor t/K\rfloor}\), Exponential: \(\displaystyle \eta_t=\eta_0e^{-kt}\)\par
\noindent Cosine Annealing: \(\displaystyle \eta_t=\eta_{\min}+\frac12(\eta_{\max}-\eta_{\min}) \left(1+\cos\frac{\pi t}{T}\right)\)\par
\noindent Warmup: \(\displaystyle \eta_t=\eta_{\mathrm{max}}\frac{t}{T_w}\)\par

\noindent \(\displaystyle \frac{\partial J}{\partial w_{u,v}}=\sum_{i,j}\delta_{i,j}x_{i+u,j+v} \quad \frac{\partial J}{\partial b}=\sum_{i,j}\delta_{i,j} \quad \frac{\partial J}{\partial x_{a,b}}=\sum_{i,j}\delta_{i,j}w_{a-i,b-j}\)\par

\subsection*{Sequence Probability Factorization}
\noindent Probability Chain Rule: \(\displaystyle P(x_{1:T})=P(x_1)\prod_{t=2}^{T}P(x_t\mid x_{1:t-1})\)\par
\noindent First-Order Markov Assumption: \(\displaystyle P(x_{1:T})=P(x_1)\prod_{t=2}^{T}P(x_t\mid x_{t-1})\)\par
\noindent Second-Order Markov Assumption: \(\displaystyle P(x_{1:T})=P(x_1,x_2)\prod_{t=3}^{T}P(x_t\mid x_{t-2},x_{t-1})\)\par
\noindent Hidden Markov Model: \(\displaystyle P(x_{1:T},h_{1:T}\mid\theta)=P(h_1)\prod_{t=2}^{T}P(h_t\mid h_{t-1})\prod_{t=1}^{T}P(x_t\mid h_t)\)\par
\noindent Transition Distribution: \(\displaystyle P(h_t\mid h_{1:t-1})=P(h_t\mid h_{t-1})\)\par
\noindent Emission Distribution: \(\displaystyle P(x_t\mid h_{1:t},x_{1:t-1})=P(x_t\mid h_t)\)\par

\subsection*{Long Short-Term Memory}
\noindent \textbf{Forget Gate}: \(\displaystyle f_t=\sigma(W_f[x_t;h_{t-1}]+b_f)\) \(\to\) Old Memory to Retain\par
\noindent \textbf{Input Gate}: \(\displaystyle i_t=\sigma(W_i[x_t;h_{t-1}]+b_i)\) \(\to\) New Info to Write\par
\noindent \textbf{Output Gate}: \(\displaystyle o_t=\sigma(W_o[x_t;h_{t-1}]+b_o)\) \(\to\) New Memory to Expose\par
\noindent \textbf{Candidate Cell State}: \(\displaystyle \tilde c_t=\tanh(W_c[x_t;h_{t-1}]+b_c)\) \(\to\) New Info\par
\noindent \textbf{Cell State Update}: \(\displaystyle c_t=f_t\odot c_{t-1}+i_t\odot\tilde c_t\) \(\to\) New Memory\par
\noindent \textbf{Hidden State}: \(\displaystyle h_t=o_t\odot\tanh(c_t)\) \(\to\) Representation\par
\noindent \(\displaystyle \left.\frac{\partial c_t}{\partial c_{t-1}}\right|_{\mathrm{direct}}=\operatorname{diag}(f_t) \to \left.\frac{\partial c_T}{\partial c_t}\right|_{\mathrm{direct}}=\prod_{k=t+1}^{T}\operatorname{diag}(f_k)\)\par
\noindent Separate Memory + Additive Update \(\to\) Better Gradients Flow\par
\noindent Parameter Count: \(\displaystyle 4H(I+H+1) + O(H + 1)\)\par
\subsection*{GRU}
\noindent \textbf{Reset Gate}: \(\displaystyle r_t=\sigma(W_r[x_t;h_{t-1}]+b_r)\) \(\to\) Old Memory to Reuse\par
\noindent \textbf{Update Gate}: \(\displaystyle z_t=\sigma(W_z[x_t;h_{t-1}]+b_z)\) \(\to\) Old Memory to Retain\par
\noindent \textbf{Candidate Hidden State}: \(\displaystyle \tilde h_t=\tanh(W_xx_t+r_t\odot(W_hh_{t-1})+b_h)\) \(\to\) New Info\par
\noindent \textbf{Hidden State Update}: \(\displaystyle h_t=z_t\odot h_{t-1}+(1-z_t)\odot\tilde h_t\) \(\to\) Representation\par
\noindent \(\displaystyle \left.\frac{\partial h_t}{\partial h_{t-1}}\right|_{\mathrm{direct}}=\operatorname{diag}(z_t)\to\left.\frac{\partial h_T}{\partial h_t}\right|_{\mathrm{direct}}=\prod_{k=t+1}^{T}\operatorname{diag}(z_k)\)\par
\noindent Unified Memory + Additive Update \(\to\) Better Gradient Flow\par
\noindent Parameter Count: \(\displaystyle 3H(I+H+1) + O(H + 1)\)\par
\subsection*{Autoregressive Generation}

\noindent \(\displaystyle L=-\log P(x_{1:T})\approx-\sum_{t=1}^{T}\log P_\theta(x_t\mid h_{t-1})=-\sum_{t=1}^{T}\log p_{t-1}[x_t]\)\par
