% 条目：名称靠左、学位或职位位于整行中央略偏右、日期靠右；长文字自动换行。
% 用法：\cventry{名称}{学位 / 职位}{日期}
% 优先使用实际文字宽度；发生碰撞时再退回可换行的宽中栏布局。
\newsavebox{\cvEntryLeft}
\newsavebox{\cvEntryMiddle}
\newsavebox{\cvEntryRight}
\newif\ifcvEntryFits
\newlength{\cvEntryOffset}
\setlength{\cvEntryOffset}{12pt} % 学位／职位相对页面中线向右偏移约 4 mm。
\newcommand{\cventry}[3]{%
  \par\addvspace{7pt}%
  \Needspace{4\baselineskip}%
  \begingroup
  \sbox{\cvEntryLeft}{\textbf{#1}}%
  \sbox{\cvEntryMiddle}{\textit{#2}}%
  \sbox{\cvEntryRight}{#3}%
  \cvEntryFitstrue
  \ifdim\dimexpr\wd\cvEntryLeft+0.5\wd\cvEntryMiddle+8pt\relax>\dimexpr0.5\linewidth+\cvEntryOffset\relax
    \cvEntryFitsfalse
  \fi
  \ifdim\dimexpr\wd\cvEntryRight+0.5\wd\cvEntryMiddle+8pt\relax>\dimexpr0.5\linewidth-\cvEntryOffset\relax
    \cvEntryFitsfalse
  \fi
  \ifcvEntryFits
    \noindent\makebox[\linewidth][l]{%
      \rlap{\usebox{\cvEntryLeft}}%
      \hspace*{\cvEntryOffset}%
      \makebox[\linewidth][c]{\usebox{\cvEntryMiddle}}%
      \hspace*{-\cvEntryOffset}%
      \llap{\usebox{\cvEntryRight}}}%
    \par\nobreak
  \else
  \noindent\begin{tabularx}{\linewidth}{@{}>{\hsize=0.84\hsize\advance\hsize by\cvEntryOffset\linewidth=\hsize\raggedright\arraybackslash}X@{\hspace{8pt}}>{\hsize=1.32\hsize\linewidth=\hsize\centering\arraybackslash}X@{\hspace{8pt}}>{\hsize=0.84\hsize\advance\hsize by-\cvEntryOffset\linewidth=\hsize\raggedleft\arraybackslash}X@{}}%
    \textbf{#1} & \textit{#2} & #3
  \end{tabularx}\par\nobreak
  \fi
  \endgroup
}
